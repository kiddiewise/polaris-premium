Legacy block-theme compatibility placeholder.

This theme is intentionally a classic PHP theme. The former templates/index.html
contained PHP markup, which block templates do not execute, and caused WordPress
to classify the theme as a block theme. The active fallback remains /index.php.

--- Former templates/index.html content (kept for reference) ---

<!-- fallback html template for block-theme compatibility -->
<!DOCTYPE html>
<html <?php language_attributes(); ?>>
<head>
  <meta charset="<?php bloginfo('charset'); ?>">
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <?php wp_head(); ?>
</head>
<body <?php body_class(); ?>>
<?php wp_body_open(); ?>
  <main id="content" class="site-content" role="main">
    <?php
    if ( have_posts() ) {
      while ( have_posts() ) {
        the_post();
        the_content();
      }
    } else {
      echo '<p>' . esc_html__("Hiç içerik yok.", 'polaris') . '</p>';
    }
    ?>
  </main>
  <?php wp_footer(); ?>
</body>
</html>
